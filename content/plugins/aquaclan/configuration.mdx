# Configuration

## Names and tags

```yaml filename="config.yml"
# Do not change this
version: 1

# tells the console when a newer version is on spigot, checked once on startup
update-notify: true

# where clan data is stored
# type: SQLITE (default), MYSQL, MARIADB or JSON
# to switch: /clan admin migrate <type>, then change type here and restart
storage:
  type: SQLITE
  # only used by MYSQL and MARIADB
  mysql:
    host: localhost
    port: 3306
    database: aquaclan
    username: root
    password: ''
    ssl: false

clan:
  # rules for clan names
  name:
    min-length: 3
    max-length: 10
    # a name must match this regex
    pattern: '^[a-zA-Z0-9]+$'
    # names nobody is allowed to use
    blacklist:
      - admin
      - staff
      - owner
      - mod
      - server
  tag:
    # color used when a clan has no custom tag
    default-color: '&#54ADF4'
    # /clan custom tag <text> - supports legacy, hex (&#RRGGBB) and MiniMessage
    custom:
      enabled: true
      permission: 'aquaclan.clan.tag.custom'
      # length limits count only visible characters, color codes are ignored
      min-length: 3
      max-length: 16
      max-raw-length: 256
      formats:
        legacy: ''
        hex: 'aquaclan.clan.tag.hex'
        bold: 'aquaclan.clan.tag.bold'
        italic: 'aquaclan.clan.tag.italic'
        underlined: 'aquaclan.clan.tag.underlined'
        strikethrough: 'aquaclan.clan.tag.strikethrough'
        obfuscated: 'aquaclan.clan.tag.obfuscated'
        gradient: 'aquaclan.clan.tag.gradient'
        rainbow: 'aquaclan.clan.tag.rainbow'
        minimessage: 'aquaclan.clan.tag.minimessage'
  # base member limit + per-permission overrides, the highest value wins
  member-limits:
    default: 4
    permissions:
      'aquaclan.limit.vip': 8
      'aquaclan.limit.mvp': 12
      'aquaclan.limit.legend': 20
  # max clan homes (level rewards can raise this)
  max-homes: 1
  # max allies, 0 = unlimited (level rewards can raise this)
  max-allies: 5
  # default friendly fire between members of the same clan
  friendly-fire: false
  chat:
    enabled: true
    format: '&#54ADF4&lCLAN &7» %clan% &8• &f%player%&7: &f%message%'
  home:
    cooldown-seconds: 5
    # delay before teleporting, cancelled if the player moves
    warmup-seconds: 3
    cancel-on-move: true
  create:
    # cooldown after creating a clan
    cooldown-seconds: 60
  disband:
    # require /clan disband confirm
    confirm: true
    # disbanding starts the create cooldown to stop create/disband abuse
    cooldown-seconds: 300
  # banner designer, opened with /clan banner edit
  banner:
    permission: 'aquaclan.clan.leader.custom.banner'
    # max pattern layers (hard cap 16)
    max-layers: 6

placeholders:
  # value of %aquaclan_has_clan%
  has-clan:
    with-clan: '&#56FF00Yes'
    without-clan: '&#FF2D2DNo'

  # text used by the %aquaclan_<x>% placeholders when the player has no clan
  # leave a value empty to show nothing at all
  no-clan:
    name: '&7No clan'
    tag: '&7No clan'
    nickname: '&7No clan'
    description: '&7No description'
    role: '&7None'
    leader: '&7None'
    created: '&7-'

# points earned in combat, they raise the clan level (see level.yml)
points:
  per-kill: 10
  per-death: 0
  # extra points for killing a rival clan member
  rival-kill-bonus: 5
  # points lost for killing an ally clan member
  ally-kill-penalty: 15

# clan invitations, they survive restarts and reach offline players
invites:
  # how long an invitation stays valid, 0 = never expires
  expire-seconds: 604800
  # max invitations a single player can hold at once, 0 = unlimited
  max-pending: 15
  # max invitations one clan can have waiting for an answer, 0 = unlimited
  max-per-clan: 20
  # tell a player about pending invitations when they join
  notify-on-join: true

# chat input used by the menus, the player writes the answer in chat
chat-input:
  # seconds the player has to answer before the request is cancelled
  timeout-seconds: 60
  # words that cancel the request
  cancel-words:
    - cancel
    - cancelar
  bossbar:
    enabled: true
    # PINK, BLUE, RED, GREEN, YELLOW, PURPLE or WHITE
    color: BLUE
    # PROGRESS, NOTCHED_6, NOTCHED_10, NOTCHED_12 or NOTCHED_20
    style: PROGRESS
  title:
    enabled: true
    fade-in: 5
    stay: 60
    fade-out: 10
  sounds:
    open: BLOCK_NOTE_BLOCK_PLING
    success: ENTITY_PLAYER_LEVELUP
    cancel: BLOCK_NOTE_BLOCK_BASS

# money sinks, needs Vault - money is only removed, never created
economy:
  enabled: false
  create-cost: 500.0
  costs:
    custom-tag: 1000.0
    rename: 750.0
    sethome: 250.0
    # charged when a player takes a copy of the clan banner
    banner-give: 250.0
    # charged when a player sends a clan invitation
    invite: 0.0

menu:
  # max clans listed by the ranking menus
  top-limit: 200
  # shown on menu buttons that need a clan when the viewer has none
  block-not-have-clan:
    enabled: true
    material: RED_STAINED_GLASS_PANE
    display_name: '&#FF2D2D&lI need to create a clan'
    lore:
      - '&7You are not in a clan yet.'
      - ''
      - '&#FFEE00Create one with &f/clan create <name>'

# worlds where the plugin works, empty = every world
worlds: []
```
